    ```tsx
    import { Component, type ErrorInfo, type ReactNode } from 'react'
    import { AlertTriangle } from 'lucide-react'
    import { useTranslation } from 'react-i18next'
    import { Button } from '@/components/ui/button'

    interface ErrorBoundaryProps {
      children: ReactNode
    }

    interface ErrorBoundaryState {
      error: Error | null
    }

    class ErrorBoundaryClass extends Component<ErrorBoundaryProps, ErrorBoundaryState> {
      state: ErrorBoundaryState = { error: null }

      static getDerivedStateFromError(error: Error): ErrorBoundaryState {
        return { error }
      }

      componentDidCatch(error: Error, info: ErrorInfo) {
        console.error('Error boundary caught:', error, info.componentStack)
      }

      render() {
        if (this.state.error) {
          return <ErrorFallback error={this.state.error} onRetry={() => this.setState({ error: null })} />
        }
        return this.props.children
      }
    }

    function ErrorFallback({ error, onRetry }: { error: Error; onRetry: () => void }) {
      const { t } = useTranslation()
      return (
        <div className="flex flex-col items-center justify-center min-h-[40vh] gap-4 text-center px-4">
          <AlertTriangle className="h-12 w-12 text-muted-foreground" />
          <h2 className="text-xl font-semibold">{t('error.title')}</h2>
          <p className="text-sm text-muted-foreground max-w-sm">{error.message}</p>
          <Button variant="default" onClick={onRetry}>
            {t('common.retry')}
          </Button>
        </div>
      )
    }

    export { ErrorBoundaryClass as ErrorBoundary }
    ```
    Note: Using a separate `ErrorFallback` function component for the UI (needs `useTranslation` hook which only works in function components). The class component handles lifecycle; the function component handles i18n UI. Both exported as `ErrorBoundary`.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ErrorBoundary → React render tree | Catches uncaught render errors from any child component |
| Skeleton states → Data | Transient visual state while Firestore snapshot loads |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-05 | Information Disclosure | ErrorBoundary.error.message in UI | accept | Error message is the native Error object message (developer-oriented). In a future phase, implement a production-mode filter that shows a generic message. |
| T-01-06 | Denial of Service | Skeleton flash on fast connections | accept | Brief skeleton render on Firestore snapshot mount is acceptable for MVP. Per RESEARCH.md minimum display time can be added later. |

## Assumptions
- All Firestore hooks (`useLists`, `useListItems`, `usePantry`, `useDocument`) return `loading: boolean` that is `true` on initial mount and `false` after first snapshot
- ErrorBoundary only catches render-phase errors, not async errors (fetch failures, Firestore rejects)
</threat_model>
